{{- /* A post's optional PDF version: the file its front matter names with `pdf: "paper.pdf"`, as a page resource, or "" for a post without one. Context: the page. Returns a dict: Resource — the PDF page resource Permalink — absolute URL, e.g. for citation_pdf_url RelPermalink — site-relative URL for links on the page Size — human-readable size, e.g. "1.2 MB" DownloadName — the file name readers get when they download it `pdf` names a file in the post's own folder. The deploy never prints a post that has one (render-post-pdfs.sh skips it), so this file is the post's only PDF: the View/Download buttons, the sidebar's PDF button, Google Scholar and Zenodo all use it. */ -}} {{- $result := "" -}} {{- with .Params.pdf -}} {{- $name := strings.TrimPrefix "./" (string .) -}} {{- $resource := $.Resources.Get $name -}} {{- if and $resource (eq $resource.MediaType.SubType "pdf") -}} {{- $bytes := len $resource.Content -}} {{- $size := printf "%d kB" (int (math.Ceil (div (float $bytes) 1000))) -}} {{- if ge $bytes 1000000 -}} {{- $size = printf "%.1f MB" (div (float $bytes) 1000000) -}} {{- end -}} {{- /* "2026-015-a-short-title.pdf" rather than the stored file name. */ -}} {{- $slug := strings.TrimRight "-" (truncate 60 "" (lower (urlize $.Title))) -}} {{- $downloadName := printf "%s.pdf" (delimit (slice (string $.Params.post_id) $slug) "-") -}} {{- $result = dict "Resource" $resource "Permalink" $resource.Permalink "RelPermalink" $resource.RelPermalink "Size" $size "DownloadName" $downloadName -}} {{- else -}} {{- warnf "%s: pdf: %q is not a PDF in the post's folder" $.File.Path $name -}} {{- end -}} {{- end -}} {{- return $result -}}
